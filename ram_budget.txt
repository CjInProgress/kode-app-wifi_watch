# The budget of internal RAM for this app, read by the SDK's tools/check_ram.sh
# (catalog rule B8):
#
#   <kode-sdk>/tools/check_ram.sh build/wifi_watch.elf ram_budget.txt
#
# It starts from the SDK's tests/target/ram_budget.txt at the pinned release,
# without the lines of what this app turns off (the network and BLE), and adds
# the app's own large statics at the end, each with its reason. The radio and
# LVGL are on: radio_stack, net_stack and lv_global are back from the SDK's file.
#
# static_max is the SDK's ceiling; an app never raises it.
static_max 368520

# Every data symbol of 2048 bytes or more in internal RAM, with the source
# file that defines it and the most it may be. A static this size that is not
# listed fails the check: a buffer belongs in PSRAM (khal_alloc_psram()), and
# the ones below are here because something forces them to be internal.
#
# symbol               source file          max bytes
